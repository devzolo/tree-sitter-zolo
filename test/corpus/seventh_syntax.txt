// Keep each user's type while numbering, pairing and grouping a roster.
struct User { name: str, team: str }
let users = [
    User { name: "Ana", team: "compiler" },
    User { name: "Bruno", team: "editor" },
    User { name: "Cris", team: "compiler" },
]

let numbered: [(int, User)] = users.enumerate()
for row in numbered {
    let (index, user) = row
    print("{index}: {user.name}")
}

let nicknames = ["ana", "bruno", "cris"]
let paired: [(User, str)] = users.zip(nicknames)
for row in paired {
    let (user, nickname) = row
    print("@{nickname}: {user.team}")
}

let by_team: #{str: [User]} = users.group_by(|user| user.team)
let entries: [(str, [User])] = by_team.entries()
let counts: #{str: int} = #{}
for row in entries {
    let (team, members) = row
    counts.set(team, members.len())
}
assert(counts.get("compiler") == 2)
assert(counts.get("editor") == 1)
print("compiler: {counts.get("compiler")}")
print("editor: {counts.get("editor")}")

let empty_users: [User] = []
let empty_numbered: [(int, User)] = empty_users.enumerate()
assert(empty_numbered.len() == 0)
